/**
 * Normalizes drawn glyph strokes to a standard 1000x1000 em-square coordinate system.
 * Preserves relative vertical positioning for proper baseline alignment.
 */
object GlyphNormalizer {

    data class NormalizationResult(
        val normalizedCommands: List<PathCommand>,
        val advanceWidth: Float,
        val leftBearing: Float,
        val baselineY: Float = 750f
    )

    /**
     * Normalizes a list of path commands to fit within a 1000x1000 em-square.
     * Trims horizontal bounds (minX to maxX) while keeping vertical coordinates
     * aligned relative to the pad height (0..1000 em-square) so baseline alignment is preserved.
     *
     * @param commands The original path commands from user drawing
     * @param drawingHeight The height of the drawing area (for baseline calculation, default 1000f)
     * @param baselineY The Y coordinate of the baseline within the drawing area (default 750f)
     */
    fun normalize(
        commands: List<PathCommand>,
        drawingHeight: Float = 1000f,
        baselineY: Float = 750f
    ): NormalizationResult {
        if (commands.isEmpty()) {
            return NormalizationResult(emptyList(), 0f, 0f, baselineY)
        }

        // Collect all points to compute bounding box
        val allPoints = mutableListOf<Pair<Float, Float>>()
        for (cmd in commands) {
            for (i in cmd.points.indices step 2) {
                if (i + 1 < cmd.points.size) {
                    allPoints.add(cmd.points[i] to cmd.points[i + 1])
                }
            }
        }

        if (allPoints.isEmpty()) {
            return NormalizationResult(emptyList(), 0f, 0f, baselineY)
        }

        // Compute horizontal bounds
        val minX = allPoints.minOf { it.first }
        val maxX = allPoints.maxOf { it.first }
        val maxY = allPoints.maxOf { it.second }

        val effHeight = if (drawingHeight > 0f) drawingHeight else maxOf(1000f, maxY)
        val scale = 1000f / effHeight

        val leftBearing = 50f
        val charWidth = (maxX - minX) * scale
        val advanceWidth = charWidth + (leftBearing * 2)
        val normalizedBaselineY = baselineY * scale

        // Transform all commands: trim X, scale Y relative to pad height
        val normalizedCommands = commands.map { cmd ->
            val normalizedPoints = cmd.points.mapIndexed { index, value ->
                if (index % 2 == 0) {
                    // X coordinate: trim horizontal bounds and add left bearing
                    (value - minX) * scale + leftBearing
                } else {
                    // Y coordinate: preserve vertical position relative to pad height (0..1000)
                    value * scale
                }
            }
            cmd.copy(points = normalizedPoints)
        }

        return NormalizationResult(
            normalizedCommands = normalizedCommands,
            advanceWidth = advanceWidth,
            leftBearing = leftBearing,
            baselineY = normalizedBaselineY
        )
    }

    /**
     * Smooths a sequence of points using quadratic BÃ©zier curves.
     * Converts raw touch points into smooth path commands.
     */
    fun smoothPath(rawPoints: List<Pair<Float, Float>>): List<PathCommand> {
        if (rawPoints.isEmpty()) return emptyList()
        if (rawPoints.size == 1) {
            return listOf(PathCommand(type = "moveTo", points = listOf(rawPoints[0].first, rawPoints[0].second)))
        }

        val commands = mutableListOf<PathCommand>()

        // Start at first point
        commands.add(PathCommand(
            type = "moveTo",
            points = listOf(rawPoints[0].first, rawPoints[0].second)
        ))

        if (rawPoints.size == 2) {
            // Just a line for two points
            commands.add(PathCommand(
                type = "lineTo",
                points = listOf(rawPoints[1].first, rawPoints[1].second)
            ))
            return commands
        }

        // For 3+ points, use quadratic curves
        var i = 1
        while (i < rawPoints.size - 1) {
            val current = rawPoints[i]
            val next = rawPoints[i + 1]

            // Control point is current, end point is midpoint to next for smoother curves
            val midX = (current.first + next.first) / 2f
            val midY = (current.second + next.second) / 2f

            commands.add(PathCommand(
                type = "quadTo",
                points = listOf(current.first, current.second, midX, midY)
            ))

            i += 1
        }

        // Finish with lineTo the final point
        val last = rawPoints.last()
        commands.add(PathCommand(
            type = "lineTo",
            points = listOf(last.first, last.second)
        ))

        return commands
    }
}

/**
 * Pure-Kotlin image processing pipeline for plain-paper handwriting scans.
 * Uses Integral Image (Summed-Area Table) for fast adaptive thresholding
 * without any native OpenCV .so dependencies.
 */
object PlainPaperScanner {

    private const val MAX_DIMENSION = 2048
    private const val MIN_CHAR_WIDTH = 6
    private const val MIN_CHAR_HEIGHT = 8
    private const val DEFAULT_WINDOW_RATIO = 1f / 24f  // ~width/24 window size

    /**
     * Binarizes a page image using adaptive Bradley-Roth / Sauvola thresholding.
     * Returns a bitmap where dark ink becomes opaque black (0xFF000000)
     * and paper/shadows become fully transparent (0x00000000).
     *
     * @param bitmap Input bitmap (color or grayscale)
     * @param sensitivity Sensitivity factor (0.05-0.25), lower = more sensitive to faint ink
     * @return ARGB_8888 bitmap with transparent background and opaque ink
     */
    fun binarizePage(bitmap: Bitmap, sensitivity: Float = 0.15f): Bitmap {
        // Downscale if too large
        val scaledBitmap = if (bitmap.width > MAX_DIMENSION || bitmap.height > MAX_DIMENSION) {
            val scale = MAX_DIMENSION.toFloat() / maxOf(bitmap.width, bitmap.height)
            Bitmap.createScaledBitmap(
                bitmap,
                (bitmap.width * scale).toInt(),
                (bitmap.height * scale).toInt(),
                true
            )
        } else {
            bitmap
        }

        val width = scaledBitmap.width
        val height = scaledBitmap.height

        // Extract grayscale luminance (0.0 - 1.0)
        val gray = IntArray(width * height)
        val pixels = IntArray(width * height)
        scaledBitmap.getPixels(pixels, 0, width, 0, 0, width, height)

        for (i in 0 until pixels.size) {
            val p = pixels[i]
            // Luminance: 0.299*R + 0.587*G + 0.114*B
            val r = Color.red(p)
            val g = Color.green(p)
            val b = Color.blue(p)
            val lum = (0.299f * r + 0.587f * g + 0.114f * b) / 255f
            gray[i] = (lum * 1000).toInt() // Scale to 0-1000 for integer math
        }

        // Build integral image (summed-area table)
        val integral = LongArray((width + 1) * (height + 1))
        for (y in 0 until height) {
            var rowSum = 0L
            val rowOffset = y * width
            val intRowOffset = (y + 1) * (width + 1)
            val intPrevRowOffset = y * (width + 1)
            for (x in 0 until width) {
                val idx = rowOffset + x
                rowSum += gray[idx].toLong()
                integral[intRowOffset + x + 1] = integral[intPrevRowOffset + x + 1] + rowSum
            }
        }

        // Adaptive thresholding
        val windowSize = maxOf(3, (width * DEFAULT_WINDOW_RATIO).toInt())
        val halfWindow = windowSize / 2
        val k = sensitivity // sensitivity factor

        val output = IntArray(width * height)
        val outBitmap = Bitmap.createBitmap(width, height, Bitmap.Config.ARGB_8888)

        for (y in 0 until height) {
            for (x in 0 until width) {
                // Window bounds
                val x1 = maxOf(0, x - halfWindow)
                val y1 = maxOf(0, y - halfWindow)
                val x2 = minOf(width - 1, x + halfWindow)
                val y2 = minOf(height - 1, y + halfWindow)

                // Get sum from integral image
                val count = ((x2 - x1 + 1) * (y2 - y1 + 1)).toLong()
                val sum = integral[(y2 + 1) * (width + 1) + (x2 + 1)]
                    - integral[y1 * (width + 1) + (x2 + 1)]
                    - integral[(y2 + 1) * (width + 1) + x1]
                    + integral[y1 * (width + 1) + x1]

                val mean = sum.toFloat() / count
                val threshold = mean * (1f - k)

                val pixelValue = gray[y * width + x].toFloat()
                if (pixelValue < threshold) {
                    // Ink: opaque black
                    output[y * width + x] = -0x1000000
                } else {
                    // Paper: transparent
                    output[y * width + x] = 0
                }
            }
        }

        outBitmap.setPixels(output, 0, width, 0, 0, width, height)
        return outBitmap
    }

    /**
     * Extracts character candidates from a binarized page using ML Kit Text Recognition.
     *
     * @param originalBitmap Original color image (for context)
     * @param binarizedBitmap Binarized transparent bitmap from binarizePage()
     * @return List of ScannedCharCandidate with alpha masks
     */
    suspend fun extractCandidates(
        originalBitmap: Bitmap,
        binarizedBitmap: Bitmap
    ): List<ScannedCharCandidate> {
        val candidates = mutableListOf<ScannedCharCandidate>()

        // ML Kit Text Recognition
        val recognizer: TextRecognizer = TextRecognition.getClient(TextRecognizerOptions.DEFAULT_OPTIONS)
        val inputImage = InputImage.fromBitmap(originalBitmap, 0)
        val result: Text = recognizer.process(inputImage).await()

        // Process each line -> element (word) -> symbol (character)
        for (block in result.textBlocks) {
            for (line in block.lines) {
                // Get line baseline for vertical alignment
                val lineBox = line.boundingBox ?: continue
                val lineBottom = lineBox.bottom
                val lineTop = lineBox.top

                for (element in line.elements) {
                    val elementText = element.text
                    val elementBox = element.boundingBox ?: continue

                    // Try to get individual symbols
                    val symbols = element.symbols

                    if (symbols.isNotEmpty()) {
                        // ML Kit provided individual character boxes
                        for (symbol in symbols) {
                            val symbolText = symbol.text
                            val symbolBox = symbol.boundingBox ?: continue

                            val crop = cropAndTrim(
                                binarizedBitmap,
                                symbolBox,
                                lineBottom
                            )
                            crop?.let { candidates.add(it) }
                        }
                    } else if (elementText.length > 1) {
                        // No symbols - split word into characters using vertical projection
                        val wordCandidates = splitWordByProjection(
                            binarizedBitmap,
                            elementBox,
                            elementText,
                            lineBottom
                        )
                        candidates.addAll(wordCandidates)
                    } else {
                        // Single character word
                        val crop = cropAndTrim(
                            binarizedBitmap,
                            elementBox,
                            lineBottom
                        )
                        crop?.let { candidates.add(it) }
                    }
                }
            }
        }

        recognizer.close()
        return candidates
    }

    /**
     * Crops a character from the binarized bitmap and trims transparent padding.
     * Preserves vertical baseline offset relative to the line's bottom.
     */
    private fun cropAndTrim(
        binarized: Bitmap,
        box: Rect,
        lineBottom: Int
    ): ScannedCharCandidate? {
        val cropRect = Rect(
            maxOf(0, box.left),
            maxOf(0, box.top),
            minOf(binarized.width, box.right),
            minOf(binarized.height, box.bottom)
        )

        if (cropRect.width() < MIN_CHAR_WIDTH || cropRect.height() < MIN_CHAR_HEIGHT) {
            return null
        }

        // Find actual ink bounds to trim excess transparent space
        val pixels = IntArray(cropRect.width() * cropRect.height())
        binarized.getPixels(pixels, 0, cropRect.width(), cropRect.left, cropRect.top, cropRect.width(), cropRect.height())

        var minX = cropRect.width()
        var maxX = 0
        var minY = cropRect.height()
        var maxY = 0
        var inkFound = false

        for (y in 0 until cropRect.height()) {
            for (x in 0 until cropRect.width()) {
                val p = pixels[y * cropRect.width() + x]
                if (Color.alpha(p) > 0) { // Opaque pixel (ink)
                    inkFound = true
                    minX = minOf(minX, x)
                    maxX = maxOf(maxX, x)
                    minY = minOf(minY, y)
                    maxY = maxOf(maxY, y)
                }
            }
        }

        if (!inkFound || minX > maxX || minY > maxY) return null

        // Trim to ink bounds with small padding
        val padding = 2
        val trimmedRect = Rect(
            maxOf(0, minX - padding),
            maxOf(0, minY - padding),
            minOf(cropRect.width(), maxX + padding + 1),
            minOf(cropRect.height(), maxY + padding + 1)
        )

        val trimmedWidth = trimmedRect.width()
        val trimmedHeight = trimmedRect.height()

        if (trimmedWidth < MIN_CHAR_WIDTH || trimmedHeight < MIN_CHAR_HEIGHT) return null

        // Create trimmed bitmap
        val trimmedPixels = IntArray(trimmedWidth * trimmedHeight)
        for (y in 0 until trimmedHeight) {
            for (x in 0 until trimmedWidth) {
                val srcX = trimmedRect.left + x
                val srcY = trimmedRect.top + y
                if (srcX < cropRect.width() && srcY < cropRect.height()) {
                    trimmedPixels[y * trimmedWidth + x] = pixels[srcY * cropRect.width() + srcX]
                }
            }
        }

        val alphaMask = Bitmap.createBitmap(trimmedWidth, trimmedHeight, Bitmap.Config.ARGB_8888)
        alphaMask.setPixels(trimmedPixels, 0, trimmedWidth, 0, 0, trimmedWidth, trimmedHeight)

        // Baseline Y relative to top of trimmed crop
        val baselineY = (lineBottom - (cropRect.top + trimmedRect.top)).toFloat()

        return ScannedCharCandidate(
            proposedChar = "", // Will be set by caller
            alphaMask = alphaMask,
            advanceWidth = trimmedWidth.toFloat(),
            baselineY = baselineY
        )
    }

    /**
     * Splits a word box into individual characters using vertical projection valley finding.
     */
    private fun splitWordByProjection(
        binarized: Bitmap,
        wordBox: Rect,
        wordText: String,
        lineBottom: Int
    ): List<ScannedCharCandidate> {
        val candidates = mutableListOf<ScannedCharCandidate>()

        // Crop word region
        val cropRect = Rect(
            maxOf(0, wordBox.left),
            maxOf(0, wordBox.top),
            minOf(binarized.width, wordBox.right),
            minOf(binarized.height, wordBox.bottom)
        )

        if (cropRect.width() < MIN_CHAR_WIDTH || cropRect.height() < MIN_CHAR_HEIGHT) {
            return candidates
        }

        val pixels = IntArray(cropRect.width() * cropRect.height())
        binarized.getPixels(pixels, 0, cropRect.width(), cropRect.left, cropRect.top, cropRect.width(), cropRect.height())

        // Vertical projection: count ink pixels per column
        val projection = IntArray(cropRect.width())
        for (x in 0 until cropRect.width()) {
            var count = 0
            for (y in 0 until cropRect.height()) {
                val p = pixels[y * cropRect.width() + x]
                if (Color.alpha(p) > 0) count++
            }
            projection[x] = count
        }

        // Find valleys (local minima in projection) - character boundaries
        val valleys = findValleys(projection)

        // Map valleys to character positions
        // Distribute characters evenly between valleys
        val charCount = wordText.length
        if (charCount == 0) return candidates

        var prevX = 0
        var charIndex = 0

        for (valleyX in valleys) {
            if (charIndex >= charCount) break

            val charBox = Rect(
                cropRect.left + prevX,
                cropRect.top,
                cropRect.left + valleyX,
                cropRect.bottom
            )

            val crop = cropAndTrim(binarized, charBox, lineBottom)
            crop?.let {
                candidates.add(it.copy(proposedChar = wordText[charIndex].toString()))
            }

            prevX = valleyX
            charIndex++
        }

        // Last character
        if (charIndex < charCount) {
            val charBox = Rect(
                cropRect.left + prevX,
                cropRect.top,
                cropRect.right,
                cropRect.bottom
            )
            val crop = cropAndTrim(binarized, charBox, lineBottom)
            crop?.let {
                candidates.add(it.copy(proposedChar = wordText[charIndex].toString()))
            }
        }

        return candidates
    }

    private fun findValleys(projection: IntArray): List<Int> {
        val valleys = mutableListOf<Int>()
        val window = maxOf(3, projection.size / 20)
        val maxVal = projection.maxOrNull() ?: 0
        val threshold = (maxVal * 0.1f).toInt()

        for (i in window until projection.size - window) {
            val isLocalMin = (i - window until i + window).all { projection[i] <= projection[it] }
            val isLow = projection[i] <= threshold
            if (isLocalMin && isLow) {
                // Avoid too-close valleys
                if (valleys.isEmpty() || i - valleys.last() > window) {
                    valleys.add(i)
                }
            }
        }

        return valleys
    }
}

object HandwritingPageRenderer {

    private const val DPI = 150
    private const val A4_WIDTH_PX = 1240
    private const val A4_HEIGHT_PX = 1754
    private const val FONT_SIZE = 36f
    private const val LINE_SPACING_DEFAULT = 1.2f

    private val fallbackTypeface = Typeface.create("serif", Typeface.ITALIC)
    private val marginPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {

    fun getPageSize(pageSize: String): Pair<Int, Int> {
        return when (pageSize) {
            "A4" -> A4_WIDTH_PX to A4_HEIGHT_PX
            "Letter" -> 1275 to 1650
            else -> A4_WIDTH_PX to A4_HEIGHT_PX

    fun paginateText(
        draft: NoteDraft,
        pageSettings: PageSettings,
        fontSize: Float = FONT_SIZE,
        glyphsByChar: Map<String, Glyph> = emptyMap()
    ): List<PageLayout> {
        val pages = mutableListOf<PageLayout>()
        val text = draft.plainText
        val lines = text.split("\n")

        val (pageWidth, pageHeight) = getPageSize(pageSettings.pageSize)
        val usableHeight = pageHeight - pageSettings.marginTop - pageSettings.marginBottom
        val lineHeight = fontSize * pageSettings.lineSpacing
        val maxLinesPerPage = (usableHeight / lineHeight).toInt().coerceAtLeast(1)

        var charIndex = 0
        var pageIndex = 0
        var lineIndexInPage = 0
        var currentLines = mutableListOf<LineLayout>()

        for ((globalLineIndex, lineText) in lines.withIndex()) {
            val lineStartChar = charIndex
            val lineEndChar = charIndex + lineText.length

            val baselineY = pageSettings.marginTop + (lineIndexInPage + 1) * lineHeight
            val glyphLayouts = layoutLineGlyphs(
                lineText = lineText,
                charStartIndex = lineStartChar,
                baselineY = baselineY,
                draft = draft,
                fontSize = fontSize,
                glyphsByChar = glyphsByChar
            )

            currentLines.add(
                LineLayout(
                    lineIndex = globalLineIndex,
                    text = lineText,
                    charStartIndex = lineStartChar,
                    charEndIndex = lineEndChar,
                    baselineY = baselineY,
                    glyphLayouts = glyphLayouts
                )
            )

            charIndex = lineEndChar + 1
            lineIndexInPage++

            if (lineIndexInPage >= maxLinesPerPage) {
                pages.add(
                    PageLayout(
                        pageIndex = pageIndex,
                        lines = currentLines.toList(),
                        charStartIndex = pages.lastOrNull()?.charEndIndex ?: 0,
                        charEndIndex = lineEndChar
                    )
                )
                pageIndex++
                lineIndexInPage = 0
                currentLines = mutableListOf()
            }
        }

        if (currentLines.isNotEmpty()) {
            pages.add(
                PageLayout(
                    pageIndex = pageIndex,
                    lines = currentLines.toList(),
                    charStartIndex = pages.lastOrNull()?.charEndIndex ?: 0,
                    charEndIndex = text.length
                )
            )
        }

        return pages.ifEmpty {
            listOf(
                PageLayout(
                    pageIndex = 0,
                    lines = emptyList(),
                    charStartIndex = 0,
                    charEndIndex = 0
                )
            )
        }
    }

    private fun layoutLineGlyphs(
        lineText: String,
        charStartIndex: Int,
        baselineY: Float,
        draft: NoteDraft,
        fontSize: Float,
        glyphsByChar: Map<String, Glyph> = emptyMap()
    ): List<GlyphLayout> {
        val glyphs = mutableListOf<GlyphLayout>()
        var x = 0f
        val y = baselineY

        for ((i, c) in lineText.withIndex()) {
            val globalCharIndex = charStartIndex + i
            val colorSpan = draft.colorSpans.find {
                globalCharIndex in it.start until it.end
            }
            val colorHex = colorSpan?.colorHex ?: "#1E3A8A"

            val charKey = c.toString()
            val glyph = glyphsByChar[charKey]?.takeIf { it.variants.isNotEmpty() && !it.isFallback }

            val variantIndex = if (glyph != null && glyph.variants.isNotEmpty()) {
                (globalCharIndex * 31) % glyph.variants.size
            } else 0

            val width = if (glyph != null && glyph.variants.isNotEmpty()) {
                val variant = glyph.variants[variantIndex]
                if (variant.vectorCommands.isNotEmpty()) {
                    (variant.advanceWidth / 1000f * fontSize).coerceIn(fontSize * 0.35f, fontSize * 1.1f)
                } else if (variant.alphaMaskBitmap != null) {
                    fontSize * (variant.bitmapWidth.toFloat() / variant.bitmapHeight.coerceAtLeast(1))
                } else {
                    fontSize * 0.6f
                }
            } else {
                fontSize * 0.6f
            }
            val height = fontSize

            glyphs.add(
                GlyphLayout(
                    char = c,
                    charIndex = globalCharIndex,
                    x = x,
                    y = y,
                    width = width,
                    height = height,
                    colorHex = colorHex,
                    variantIndex = variantIndex,
                    glyph = glyph,
                    isFallback = glyph == null || glyph.variants.isEmpty()
                )
            )

            x += width
        }

        return glyphs
    }

    fun renderPageToBitmap(
        page: PageLayout,
        draft: NoteDraft,
        glyphsByChar: Map<String, Glyph>,
        pageSettings: PageSettings,
        fontSize: Float = FONT_SIZE,
        showRedMargins: Boolean = pageSettings.showRedMargins,
        showPageNumbers: Boolean = pageSettings.showPageNumbers,
        showRuledLines: Boolean = true
    ): Bitmap {
        val (width, height) = getPageSize(pageSettings.pageSize)
        val bitmap = Bitmap.createBitmap(width, height, Bitmap.Config.ARGB_8888)
        val canvas = Canvas(bitmap)

        canvas.drawColor(Color.WHITE)

        if (showRuledLines) {
            drawRuledLines(canvas, page, pageSettings, fontSize)
        }

        if (showRedMargins) {
            drawRedMargins(canvas, pageSettings)
        }

        for (line in page.lines) {
            for (glyphLayout in line.glyphLayouts) {
                drawGlyph(canvas, glyphLayout, glyphsByChar, fontSize)
            }
        }

        if (showPageNumbers) {
            drawPageNumber(canvas, page.pageIndex + 1, pageSettings)
        }

        return bitmap
    }

    private fun drawRuledLines(
        canvas: Canvas,
        page: PageLayout,
        pageSettings: PageSettings,
        fontSize: Float
    ) {
        val lineHeight = fontSize * pageSettings.lineSpacing
        for (line in page.lines) {
            val y = line.baselineY + fontSize * 0.2f
            canvas.drawLine(
                pageSettings.marginLeft,
                y,
                (getPageSize(pageSettings.pageSize).first - pageSettings.marginRight).toFloat(),
                y,
                ruledLinePaint
            )
        }
    }

    private fun drawRedMargins(canvas: Canvas, pageSettings: PageSettings) {
        val (width, height) = getPageSize(pageSettings.pageSize)

        val topMarginY = pageSettings.marginTop + pageSettings.redMarginOffset
        canvas.drawLine(
            pageSettings.marginLeft,
            topMarginY,
            width - pageSettings.marginRight,
            topMarginY,
            marginPaint
        )

        canvas.drawLine(
            pageSettings.marginLeft,
            pageSettings.marginTop,
            pageSettings.marginLeft,
            height - pageSettings.marginBottom,
            marginPaint
        )
    }

    private fun drawPageNumber(canvas: Canvas, pageNum: Int, pageSettings: PageSettings) {
        val (width, height) = getPageSize(pageSettings.pageSize)
        canvas.drawText(
            "Page $pageNum",
            width - pageSettings.marginRight,
            height - pageSettings.marginBottom + 20f,
            pageNumberPaint
        )
    }

    private fun drawGlyph(
        canvas: Canvas,
        glyphLayout: GlyphLayout,
        glyphsByChar: Map<String, Glyph>,
        fontSize: Float
    ) {
        val color = Color.parseColor(glyphLayout.colorHex)

        if (glyphLayout.glyph != null && glyphLayout.glyph.variants.isNotEmpty()) {
            val variant = glyphLayout.glyph.variants.getOrNull(glyphLayout.variantIndex)
                ?: glyphLayout.glyph.variants.firstOrNull()
            if (variant?.vectorCommands?.isNotEmpty() == true) {
                drawVectorGlyph(canvas, variant, glyphLayout, color, fontSize)
                return
            }
            if (variant?.alphaMaskBitmap != null) {
                drawBitmapGlyph(canvas, variant, glyphLayout, color, fontSize)
                return
            }
        }

        drawFallbackGlyph(canvas, glyphLayout.char, glyphLayout, color, fontSize)
    }

    private fun drawVectorGlyph(
        canvas: Canvas,
        variant: com.app.inktwin.core.glyph.GlyphVariant,
        glyphLayout: GlyphLayout,
        color: Int,
        fontSize: Float
    ) {
        val path = android.graphics.Path()
        val scale = fontSize / 1000f

        for (cmd in variant.vectorCommands) {
            when (cmd.type) {
                "moveTo" -> path.moveTo(cmd.points[0] * scale + glyphLayout.x, cmd.points[1] * scale + glyphLayout.y - fontSize)
                "lineTo" -> path.lineTo(cmd.points[0] * scale + glyphLayout.x, cmd.points[1] * scale + glyphLayout.y - fontSize)
                "quadTo" -> path.quadTo(
                    cmd.points[0] * scale + glyphLayout.x,
                    cmd.points[1] * scale + glyphLayout.y - fontSize,
                    cmd.points[2] * scale + glyphLayout.x,
                    cmd.points[3] * scale + glyphLayout.y - fontSize
                )
                "cubicTo" -> path.cubicTo(
                    cmd.points[0] * scale + glyphLayout.x,
                    cmd.points[1] * scale + glyphLayout.y - fontSize,
                    cmd.points[2] * scale + glyphLayout.x,
                    cmd.points[3] * scale + glyphLayout.y - fontSize,
                    cmd.points[4] * scale + glyphLayout.x,
                    cmd.points[5] * scale + glyphLayout.y - fontSize
                )
                "close" -> path.close()
            }
        }

        val paint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
            this.color = color
            style = Paint.Style.STROKE
            strokeCap = Paint.Cap.ROUND
            strokeJoin = Paint.Join.ROUND
            strokeWidth = fontSize * 0.08f
        }
        canvas.drawPath(path, paint)
    }

    private fun drawBitmapGlyph(
        canvas: Canvas,
        variant: com.app.inktwin.core.glyph.GlyphVariant,
        glyphLayout: GlyphLayout,
        color: Int,
        fontSize: Float
    ) {
        val alphaMask = variant.alphaMaskBitmap ?: return
        val drawHeight = fontSize * 1.15f
        val scale = drawHeight / alphaMask.height

        val paint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
            colorFilter = PorterDuffColorFilter(color, PorterDuff.Mode.SRC_IN)
        }

        val dstRect = RectF(
            glyphLayout.x,
            glyphLayout.y - fontSize,
            glyphLayout.x + alphaMask.width * scale,
            glyphLayout.y
        )

        canvas.drawBitmap(alphaMask, null, dstRect, paint)
    }

    private fun drawFallbackGlyph(
        canvas: Canvas,
        char: Char,
        glyphLayout: GlyphLayout,
        color: Int,
        fontSize: Float
    ) {
        val paint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
            this.color = color
            textSize = fontSize
            typeface = fallbackTypeface
        }

        canvas.drawText(char.toString(), glyphLayout.x, glyphLayout.y, paint)
    }

                }

                Glyph(

                    variants = variants,
                    isFallback = gv.glyph.isFallback
                )
            }
        }
    }

        database.glyphDao().deleteByProfileId(profileId)
        
        // Insert new glyphs and variants
        for (glyph in glyphs) {
            val glyphEntity = GlyphEntity(
                profileId = profileId,
                character = glyph.character,
                unicode = glyph.unicode,
                isFallback = glyph.isFallback
            )
            val glyphId = database.glyphDao().insert(glyphEntity)
            
            val variantEntities = glyph.variants.mapIndexed { index, variant ->
                GlyphVariantEntity(
                    glyphId = glyphId,
                    variantIndex = index,
                    vectorCommandsJson = if (variant.vectorCommands.isNotEmpty()) {
                        pathListConverter.toJson(variant.vectorCommands)
                    } else null,
                    alphaMaskBlob = bitmapToBlob(variant.alphaMaskBitmap),
                    bitmapWidth = variant.bitmapWidth,
                    bitmapHeight = variant.bitmapHeight,
                    advanceWidth = variant.advanceWidth,
                    leftBearing = variant.leftBearing,
                    topBearing = variant.topBearing,
                    ascent = variant.ascent,
                    descent = variant.descent
                )
            }
            database.glyphVariantDao().insertAll(variantEntities)
        }
    }

    override suspend fun deleteGlyphs(profileId: Long) {
        database.glyphDao().deleteByProfileId(profileId)

        }

    fun toBase64(bitmap: Bitmap?): String? {
        return bitmap?.let { bmp ->
            val stream = ByteArrayOutputStream()
            bmp.compress(Bitmap.CompressFormat.PNG, 100, stream)
            Base64.encodeToString(stream.toByteArray(), Base64.DEFAULT)
        }
    }

        ActivityResultContracts.GetMultipleContents()
    ) { uris: List<Uri> ->
        if (uris.isNotEmpty()) {
            // Convert URIs to Bitmaps on background thread
            viewModel.viewModelScope.launch(Dispatchers.IO) {
                val bitmaps = mutableListOf<Bitmap>()
                for (uri in uris) {
                    try {
                        val inputStream = context.contentResolver.openInputStream(uri)
                        inputStream?.use { stream ->
                            val bitmap = android.graphics.BitmapFactory.decodeStream(stream)
                            bitmap?.let { bitmaps.add(it) }
                        }
                    } catch (e: Exception) {
                        // Ignore failed images
                    }
                }
                if (bitmaps.isNotEmpty()) {
                    viewModel.addScannedPages(bitmaps)
                }
            }
        }
    }

    // Camera launcher
    val cameraLauncher = rememberLauncherForActivityResult(
        ActivityResultContracts.TakePicturePreview()
    ) { bitmap: Bitmap? ->
        bitmap?.let { viewModel.addScannedPages(listOf(it)) }
    }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(MaterialTheme.colorScheme.background)
    ) {

            modifier = Modifier
                .fillMaxWidth()
                .padding(16.dp),
            horizontalArrangement = Arrangement.spacedBy(8.dp),
            verticalAlignment = Alignment.CenterVertically
        ) {
            Button(
                onClick = { viewModel.setModeSwitch("draw") },
                colors = ButtonDefaults.buttonColors(
                    containerColor = if (modeSwitch == "draw") {
                        MaterialTheme.colorScheme.primary
                    } else {
                        MaterialTheme.colorScheme.surfaceVariant
                    }
                ),

            ) {

                    } else {

        for (stroke in strokes) {
            if (stroke.isEmpty()) continue
            commands.add(PathCommand(type = "moveTo", points = listOf(stroke[0].x, stroke[0].y)))
            for (i in 1 until stroke.size) {
                commands.add(PathCommand(type = "lineTo", points = listOf(stroke[i].x, stroke[i].y)))
            }
        }
        return commands
    }

@HiltViewModel
class EditorViewModel @Inject constructor(
    private val noteDraftRepository: NoteDraftRepository,
    profileRepository: com.app.inktwin.data.repository.HandwritingProfileRepository
) : ViewModel() {

    private val _textFieldValue = MutableStateFlow(TextFieldValue(""))
    val textFieldValue: StateFlow<TextFieldValue> = _textFieldValue.asStateFlow()

    private val _colorSpans = MutableStateFlow<List<ColorSpan>>(emptyList())
    val colorSpans: StateFlow<List<ColorSpan>> = _colorSpans.asStateFlow()

    private val _activeColorHex = MutableStateFlow("#1E3A8A")
    val activeColorHex: StateFlow<String> = _activeColorHex.asStateFlow()

    private val _undoStack = MutableStateFlow<List<Pair<String, List<ColorSpan>>>>(emptyList())
    private val _redoStack = MutableStateFlow<List<Pair<String, List<ColorSpan>>>>(emptyList())

    private val profileIdFlow = profileRepository.observeActiveProfile()
        .map { it?.id }
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), null)

    val canUndo: StateFlow<Boolean> = _undoStack.map { it.isNotEmpty() }
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), false)

    val canRedo: StateFlow<Boolean> = _redoStack.map { it.isNotEmpty() }
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), false)

    val characterCount: Int
        get() = _textFieldValue.value.text.count { !it.isWhitespace() }

    val wordCount: Int
        get() = _textFieldValue.value.text.trim().takeIf { it.isNotEmpty() }
            ?.split(Regex("\\s+"))?.size ?: 0

    init {
        loadDraft()
        setupAutoSave()
    }

    private fun loadDraft() {
        viewModelScope.launch {
            val profileId = profileIdFlow.value
            val drafts = noteDraftRepository.observeDrafts(profileId).firstOrNull()
            val draft = drafts?.firstOrNull()

            if (draft != null && draft.plainText.isNotEmpty()) {
                _textFieldValue.value = TextFieldValue(draft.plainText)
                _colorSpans.value = draft.colorSpans
                _activeColorHex.value = draft.colorSpans.firstOrNull()?.colorHex ?: "#1E3A8A"
            }
        }
    }

    @OptIn(FlowPreview::class)
    private fun setupAutoSave() {
        viewModelScope.launch {
            _textFieldValue
                .debounce(800)
                .distinctUntilChanged()
                .collect { saveCurrentDraft() }
        }
    }

    fun setActiveColorHex(hex: String) {
        _activeColorHex.value = hex
    }

    private fun pushUndo() {
        val snapshot = _textFieldValue.value.text to _colorSpans.value
        _undoStack.update { it + snapshot }
        _redoStack.value = emptyList()
    }

    fun undo() {
        val undoList = _undoStack.value
        if (undoList.isEmpty()) return

        val current = _textFieldValue.value.text to _colorSpans.value
        val last = undoList.last()

        _textFieldValue.value = TextFieldValue(last.first)
        _colorSpans.value = last.second

        _undoStack.value = undoList.dropLast(1)
        _redoStack.update { it + current }
    }

    fun redo() {
        val redoList = _redoStack.value
        if (redoList.isEmpty()) return

        val current = _textFieldValue.value.text to _colorSpans.value
        val next = redoList.last()

        _textFieldValue.value = TextFieldValue(next.first)
        _colorSpans.value = next.second

        _redoStack.value = redoList.dropLast(1)
        _undoStack.update { it + current }
    }

    fun onTextFieldValueChange(newValue: TextFieldValue) {
        _textFieldValue.value = newValue
    }

    fun applyColorToSelection() {
        pushUndo()
        val sel = _textFieldValue.value.selection
        if (sel.length > 0) {
            val text = _textFieldValue.value.text
            val newSpan = ColorSpan(
                start = sel.min.coerceIn(0, text.length),
                end = sel.max.coerceIn(0, text.length),
                colorHex = _activeColorHex.value,
                granularity = "selection"
            )
            _colorSpans.value = applySpanToList(newSpan, _colorSpans.value)
        } else {
            applyColorToCurrentWord()
        }
    }

    fun applyColorToCurrentWord() {
        pushUndo()
        val text = _textFieldValue.value.text
        val cursor = _textFieldValue.value.selection.start

        var start = cursor
        while (start > 0 && !text[start - 1].isWhitespace()) start--
        var end = cursor
        while (end < text.length && !text[end].isWhitespace()) end++

        if (start < end) {
            val newSpan = ColorSpan(start, end, _activeColorHex.value, "word")
            _colorSpans.value = applySpanToList(newSpan, _colorSpans.value)
        }
    }

    fun applyColorToCurrentLine() {
        pushUndo()
        val text = _textFieldValue.value.text
        val cursor = _textFieldValue.value.selection.start

        var start = cursor
        while (start > 0 && text[start - 1] != '\n') start--
        var end = cursor
        while (end < text.length && text[end] != '\n') end++

        if (start < end) {
            val newSpan = ColorSpan(start, end, _activeColorHex.value, "line")
            _colorSpans.value = applySpanToList(newSpan, _colorSpans.value)
        }
    }

    fun applyColorToCurrentParagraph() {
        pushUndo()
        val text = _textFieldValue.value.text
        val cursor = _textFieldValue.value.selection.start

        var start = cursor
        while (start > 0) {
            if (start >= 2 && text[start-1] == '\n' && text[start-2] == '\n') break
            start--
        }
        var end = cursor
        while (end < text.length) {
            if (end <= text.length - 2 && text[end] == '\n' && text[end+1] == '\n') break
            end++
        }

        if (start < end) {
            val newSpan = ColorSpan(start, end, _activeColorHex.value, "paragraph")
            _colorSpans.value = applySpanToList(newSpan, _colorSpans.value)
        }
    }

    fun clearSelectionColor() {
        pushUndo()
        val sel = _textFieldValue.value.selection
        if (sel.length > 0) {
            _colorSpans.value = _colorSpans.value.filterNot { it.start < sel.max && it.end > sel.min }
        }
    }

    private fun applySpanToList(newSpan: ColorSpan, existing: List<ColorSpan>): List<ColorSpan> {
        val nonOverlapping = existing.filter { !(it.start < newSpan.end && it.end > newSpan.start) }
        return (nonOverlapping + newSpan).sortedBy { it.start }
    }

    private suspend fun saveCurrentDraft() {
        val text = _textFieldValue.value.text
        val draft = NoteDraft(
            title = text.lines().firstOrNull()?.take(30) ?: "Untitled",
            plainText = text,
            colorSpans = _colorSpans.value,
            profileId = profileIdFlow.value
        )
        noteDraftRepository.createDraft(draft)
    }
}

@HiltViewModel
class PreviewViewModel @Inject constructor(
    private val noteDraftRepository: NoteDraftRepository,
    private val glyphRepository: GlyphRepository,
    private val handwritingProfileRepository: HandwritingProfileRepository
) : ViewModel() {

    private val _uiState = MutableStateFlow(PreviewUiState())
    val uiState: StateFlow<PreviewUiState> = _uiState.asStateFlow()

    val currentPageIndex: StateFlow<Int> = _uiState.map { state -> state.currentPageIndex }
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), 0)

    val pageBitmaps: StateFlow<List<Bitmap>> = _uiState.map { state -> state.pageBitmaps }
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())

    val totalPages: Int
        get() = _uiState.value.pages.size

    init {
        observeProfileAndGlyphs()
        loadDraft()
    }

    @OptIn(ExperimentalCoroutinesApi::class)
    private fun observeProfileAndGlyphs() {
        viewModelScope.launch {
            handwritingProfileRepository.observeActiveProfile()
                .flatMapLatest { profile ->
                    if (profile != null) {
                        glyphRepository.observeGlyphs(profile.id)
                    } else kotlinx.coroutines.flow.flowOf(emptyList())
                }
                .collect { glyphs ->
                    val glyphMap = glyphs
                        .filter { it.variants.isNotEmpty() && !it.isFallback }
                        .associateBy { it.character }
                    _uiState.update { it.copy(glyphs = glyphMap) }
                    _uiState.value.draft?.let { draft ->
                        renderPages(draft, glyphMap)
                    }
                }
        }
    }

    private fun loadDraft() {
        viewModelScope.launch {
            noteDraftRepository.observeDrafts(null)
                .firstOrNull()
                ?.firstOrNull()
                ?.let { draft ->
                    _uiState.update { it.copy(draft = draft) }
                    val glyphs = _uiState.value.glyphs
                    if (glyphs.isNotEmpty()) {
                        renderPages(draft, glyphs)
                    }
                }
        }
    }

    fun setPageSettings(settings: PageSettings) {
        _uiState.update { it.copy(pageSettings = settings) }
        val draft = _uiState.value.draft
        if (draft != null) {
            viewModelScope.launch {
                renderPages(draft, _uiState.value.glyphs)
            }
        }
    }

    fun setFontSize(size: Float) {
        _uiState.update { it.copy(fontSize = size) }
        val draft = _uiState.value.draft
        if (draft != null) {
            viewModelScope.launch {
                renderPages(draft, _uiState.value.glyphs)
            }
        }
    }

    fun setRuledLines(enabled: Boolean) {
        _uiState.update { it.copy(showRuledLines = enabled) }
        val draft = _uiState.value.draft
        if (draft != null) {
            viewModelScope.launch {
                renderPages(draft, _uiState.value.glyphs)
            }
        }
    }

    fun navigateToPage(index: Int) {
        val maxIndex = _uiState.value.pages.size - 1
        _uiState.update {
            it.copy(currentPageIndex = index.coerceIn(0, maxIndex))
        }
    }

    private suspend fun renderPages(draft: NoteDraft, glyphs: Map<String, Glyph>) {
        val pages = HandwritingPageRenderer.paginateText(
            draft = draft,
            pageSettings = _uiState.value.pageSettings,
            fontSize = _uiState.value.fontSize,
            glyphsByChar = glyphs
        )

        _uiState.update { it.copy(pages = pages) }

        val bitmaps = pages.map { page ->
            HandwritingPageRenderer.renderPageToBitmap(
                page = page,
                draft = draft,
                glyphsByChar = glyphs,
                pageSettings = _uiState.value.pageSettings,
                fontSize = _uiState.value.fontSize,
                showRedMargins = _uiState.value.pageSettings.showRedMargins,
                showPageNumbers = _uiState.value.pageSettings.showPageNumbers,
                showRuledLines = _uiState.value.showRuledLines
            )
        }

        _uiState.update { it.copy(pageBitmaps = bitmaps) }
    }

    fun exportDocument(context: Context, format: ExportFormat, scope: ExportScope) {
        viewModelScope.launch {
            _uiState.update { it.copy(isExporting = true, exportStatus = null) }

            try {
                val state = _uiState.value
                val draft = state.draft ?: return@launch

                val pagesToExport = when (scope) {
                    ExportScope.CURRENT_PAGE -> listOf(state.pages[state.currentPageIndex])
                    ExportScope.ENTIRE_DOCUMENT -> state.pages
                }

                val bitmapsToExport = when (scope) {
                    ExportScope.CURRENT_PAGE -> listOf(state.pageBitmaps[state.currentPageIndex])
                    ExportScope.ENTIRE_DOCUMENT -> state.pageBitmaps
                }

                when (format) {
                    ExportFormat.PNG -> exportAsPng(context, bitmapsToExport, draft.title)
                    ExportFormat.PDF -> exportAsPdf(context, pagesToExport, draft, state)
                }

                _uiState.update {
                    it.copy(
                        isExporting = false,
                        exportStatus = "Export successful! Check your Downloads folder."
                    )
                }
            } catch (e: Exception) {
                _uiState.update {
                    it.copy(
                        isExporting = false,
                        exportStatus = "Export failed: ${e.message}"
                    )
                }
            }
        }
    }

    private fun exportAsPng(context: Context, bitmaps: List<Bitmap>, title: String) {
        val downloadsDir = context.getExternalFilesDir(null)
        val inktwinDir = File(downloadsDir, "InkTwin").apply { mkdirs() }

        bitmaps.forEachIndexed { index, bitmap ->
            val filename = if (bitmaps.size == 1) {
                "${title.replace(Regex("[^a-zA-Z0-9]"), "_")}.png"
            } else {
                "${title.replace(Regex("[^a-zA-Z0-9]"), "_")}_page${index + 1}.png"
            }
            val file = File(inktwinDir, filename)
            FileOutputStream(file).use { out ->
                bitmap.compress(Bitmap.CompressFormat.PNG, 100, out)
            }
        }
    }

    private fun exportAsPdf(context: Context, pages: List<PageLayout>, draft: NoteDraft, state: PreviewUiState) {
        val document = PdfDocument()
        val (pageWidth, pageHeight) = HandwritingPageRenderer.getPageSize(state.pageSettings.pageSize)

        for ((index, page) in pages.withIndex()) {
            val pageInfo = PdfDocument.PageInfo.Builder(pageWidth, pageHeight, index + 1).create()
            val pdfPage = document.startPage(pageInfo)

            val bitmap = HandwritingPageRenderer.renderPageToBitmap(
                page = page,
                draft = draft,
                glyphsByChar = state.glyphs,
                pageSettings = state.pageSettings,
                fontSize = state.fontSize,
                showRedMargins = state.pageSettings.showRedMargins,
                showPageNumbers = state.pageSettings.showPageNumbers,
                showRuledLines = state.showRuledLines
            )

            pdfPage.canvas.drawBitmap(bitmap, 0f, 0f, null)
            document.finishPage(pdfPage)
            bitmap.recycle()
        }

        val downloadsDir = context.getExternalFilesDir(null)
        val inktwinDir = File(downloadsDir, "InkTwin").apply { mkdirs() }
        val filename = "${draft.title.replace(Regex("[^a-zA-Z0-9]"), "_")}.pdf"
        val file = File(inktwinDir, filename)

        FileOutputStream(file).use { out ->
            document.writeTo(out)
        }
        document.close()
    }

    fun clearExportStatus() {
        _uiState.update { it.copy(exportStatus = null) }
    }
}
